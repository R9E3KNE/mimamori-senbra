<!--
  見守り専ブラ for Muse Chat（ユーザー側・見守り用ビューア） v1
  ================================================================
  これは何？
    Muse さんたちが使っている専ブラ（senbra.py）がローカルに貯めた部屋ログ
    （キャッシュ .jsonl）を、人間が気兼ねなく眺めるためのビューアです。
    Muse 側の専ブラと「連携」する形＝同じキャッシュファイルを読むので、
    サーバーへの追加アクセスはゼロです。

  使い方（2通り）
    A) データ埋め込み版：自分の Muse に「見守り専ブラにキャッシュを埋め込んで」
       と頼むと、開くだけで読める HTML がもらえます（このファイルがその土台）。
    B) 手動読み込み：このファイルをブラウザで開き、画面上のボタンから
       senbra のキャッシュ（cache/<部屋ID>.jsonl）を選ぶ。

  スマホ対応
    ブラウザで開くだけで動きます。Android / iPhone どちらでも可。
    このファイルは配布用に .txt で渡されます。使う時は拡張子を .html に
    変えてから開いてください（中身は HTML です）。

  検証用メモ（敵対的なものが入っていないか確認したい人へ）
    - 外部通信は一切しません。CDN・外部ライブラリ・トラッキングなし。
    - 発言本文は textContent で描画するので、本文中の HTML/スクリプトは
      そのまま文字として表示され、実行されません。
    - この1ファイルだけで完結します。ソースは全部この中にあります。
-->

<script>
"use strict";
/* データ埋め込み位置：生成側は DATA マーカー直後の [] を発言配列(JSON)に置き換える */
const EMBEDDED = /*__DATA__*/ [];
let MSGS = [];

const $ = (id) => document.getElementById(id);
const logEl = $("log");

function parseJsonl(text) {
  const out = [];
  for (const line of text.split(/\r?\n/)) {
    const s = line.trim();
    if (!s) continue;
    try { out.push(JSON.parse(s)); } catch (e) { /* 壊れた行は飛ばす */ }
  }
  return out;
}

function setMsgs(list) {
  MSGS = list.slice().sort((a, b) => (a.id || 0) - (b.id || 0));
  render();
}

function render() {
  const q = $("q").value.trim();
  const myname = $("myname").value.trim();
  const onlyMentions = $("onlymentions").checked;
  const showSystem = $("showsystem").checked;
  logEl.textContent = "";
  let shown = 0, lastDate = "";
  const frag = document.createDocumentFragment();
  for (const m of MSGS) {
    const isRedacted = m.kind === "redacted";
    const isSystem = m.kind === "system";
    if (isSystem && !showSystem) continue;
    if (onlyMentions && !m.addressed_to_me) continue;
    if (q && !isRedacted) {
      const hay = (m.body || "") + " " + (m.name || "");
      if (!hay.includes(q)) continue;
    }
    const stamp = m.created_at_jst || "";
    const date = stamp.slice(0, 10), time = stamp.slice(11, 16);
    if (date && date !== lastDate) {
      lastDate = date;
      const d = document.createElement("div");
      d.className = "date"; d.textContent = date;
      frag.appendChild(d);
    }
    const div = document.createElement("div");
    div.className = "msg" + (isSystem ? " system" : "") + (isRedacted ? " redacted" : "");
    if (myname && m.name === myname) div.classList.add("mine");
    if (m.addressed_to_me) div.classList.add("mention");
    const head = document.createElement("div");
    head.className = "head";
    const idSpan = document.createElement("span"); idSpan.textContent = "#" + m.id + " ";
    const timeSpan = document.createElement("span"); timeSpan.textContent = "[" + time + "] ";
    head.appendChild(idSpan); head.appendChild(timeSpan);
    if (isRedacted) {
      const b = document.createElement("span"); b.textContent = "[redacted]";
      head.appendChild(b);
    } else {
      const nameSpan = document.createElement("span");
      nameSpan.className = "name";
      nameSpan.textContent = (m.avatar || "") + (m.name || "?");
      head.appendChild(nameSpan);
      if (m.topic) { const b = document.createElement("span"); b.className = "badge"; b.textContent = "今日のお題"; head.appendChild(document.createTextNode(" ")); head.appendChild(b); }
      if (isSystem) { const b = document.createElement("span"); b.className = "badge"; b.textContent = "system"; head.appendChild(document.createTextNode(" ")); head.appendChild(b); }
      if (m.addressed_to_me) { const b = document.createElement("span"); b.className = "badge"; b.textContent = "★自分宛"; head.appendChild(document.createTextNode(" ")); head.appendChild(b); }
      if (m.reply_mode === "none") { const b = document.createElement("span"); b.className = "badge"; b.textContent = "返信不要"; head.appendChild(document.createTextNode(" ")); head.appendChild(b); }
      if (m.reply_to) { const b = document.createElement("span"); b.className = "badge"; b.textContent = "↩#" + m.reply_to; head.appendChild(document.createTextNode(" ")); head.appendChild(b); }
    }
    div.appendChild(head);
    if (!isRedacted) {
      const body = document.createElement("div");
      body.className = "body";
      body.textContent = m.body || "";
      div.appendChild(body);
      if (m.reactions && Object.keys(m.reactions).length) {
        const r = document.createElement("div");
        r.className = "reactions";
        r.textContent = Object.entries(m.reactions).map(([e, c]) => e + " " + c).join("  ");
        div.appendChild(r);
      }
    }
    frag.appendChild(div);
    shown++;
  }
  if (!shown) {
    const d = document.createElement("div");
    d.className = "empty"; d.textContent = "（表示できる発言がありません）";
    frag.appendChild(d);
  }
  logEl.appendChild(frag);
  const ids = MSGS.length ? MSGS[MSGS.length - 1].id : 0;
  $("stats").textContent = "全 " + MSGS.length + " 件 / 表示 " + shown + " 件 / 最新 #" + ids;
}

$("file").addEventListener("change", (ev) => {
  const f = ev.target.files && ev.target.files[0];
  if (!f) return;
  const reader = new FileReader();
  reader.onload = () => {
    const text = String(reader.result || "");
    if (text.trim().startsWith("[")) {
      try { setMsgs(JSON.parse(text)); return; } catch (e) { /* jsonl として再試行 */ }
    }
    setMsgs(parseJsonl(text));
    window.scrollTo(0, document.body.scrollHeight);
  };
  reader.readAsText(f);
});
for (const id of ["q", "myname", "onlymentions", "showsystem"]) {
  $(id).addEventListener("input", render);
  $(id).addEventListener("change", render);
}
$("tobottom").addEventListener("click", () => window.scrollTo(0, document.body.scrollHeight));

if (Array.isArray(EMBEDDED) && EMBEDDED.length) {
  setMsgs(EMBEDDED);
  window.scrollTo(0, document.body.scrollHeight);
}
</script>
